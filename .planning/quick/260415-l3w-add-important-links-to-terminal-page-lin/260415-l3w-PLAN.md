3. **Replace formatTwoColumn** with a JSX-returning `renderTwoColumn` function. Since "kaleb.nim@gmail.com" is 19 chars (exceeds current padEnd(18)), increase left value padEnd to 22:

```tsx
const renderTwoColumn = (row: StatusRow): React.ReactNode => {
  const leftLabel = `  ${row.left.label.padEnd(22, ' ')}: `;
  const rightLabel = `${row.right.label.padEnd(25, ' ')}: `;

  const renderCell = (cell: StatusCell, padWidth: number, isLeft: boolean) => {
    const displayValue = isLeft ? cell.value.padEnd(padWidth, ' ') : cell.value;
    if (cell.href) {
      return (
        <a
          href={cell.href}
          className={styles.goldLink}
          target={cell.external ? "_blank" : undefined}
          rel={cell.external ? "noopener noreferrer" : undefined}
        >
          {displayValue}
        </a>
      );
    }
    return displayValue;
  };

  return (
    <>
      {leftLabel}
      {renderCell(row.left, 22, true)}
      {rightLabel}
      {renderCell(row.right, 0, false)}
    </>
  );
};
```

IMPORTANT: The padding spaces go INSIDE the anchor tag for link cells (so the `<a>` covers the padded area — this is correct since `<a>` inherits monospace font from `<pre>`). Non-link cells just render the padded string directly.

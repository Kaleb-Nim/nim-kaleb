```typescript
import { appendFileSync, mkdirSync } from 'node:fs';
import { join } from 'node:path';

const LOG_DIR = process.env.LOG_DIR ?? '/var/log/kaleb-voice';

export interface TurnLog {
  ts: number;                    // Date.now() epoch ms
  sessionId: string;
  role: 'user' | 'assistant';
  text: string;
  latency: {
    asrMs: number | null;        // speech_started → onTranscriptFinal
    llmTtftMs: number | null;    // streamLlmResponse call → first onChunk
    ttsTtfaMs: number | null;    // TTS WS open → first onAudioDelta
    totalMs: number | null;      // turnStart → onDone
  };
}

export function initLogDir(): void {
  mkdirSync(LOG_DIR, { recursive: true });
}

export function logTurn(entry: TurnLog): void {
  Promise.resolve()
    .then(() => {
      appendFileSync(join(LOG_DIR, `${entry.sessionId}.ndjson`), JSON.stringify(entry) + '\n');
    })
    .catch((err) => {
      console.error('[logger] write failed:', err);
    });
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Visitor audio -> ASR transcript | Visitor speech is transcribed and logged as text — contains visitor utterances |
| Log files on ECS filesystem | NDJSON files contain session transcripts, accessible only via SSH/container console |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-06-01 | Information Disclosure | logger.ts — transcript text in log files | accept | Transcripts are PII-adjacent but not linkable to identity (no auth, no IP logging). Files only accessible via SSH. Portfolio-scale risk is low. |
| T-06-02 | Denial of Service | logger.ts — appendFileSync could fill disk | accept | Portfolio traffic is minimal (single concurrent user). Log files are small (one line per turn). No rotation needed at this scale. |
| T-06-03 | Tampering | logger.ts — LOG_DIR env var injection | mitigate | `LOG_DIR` is read from server environment only (not from browser messages). Path is used in `join()` with session UUID filename — no path traversal possible since `sessionId` is `crypto.randomUUID()` (no user-controlled path components). |
| T-06-04 | Information Disclosure | logger.ts — log path served by HTTP | mitigate | `/health` endpoint returns plain text only. No static file serving configured in `index.ts`. LOG_DIR defaults to `/var/log/kaleb-voice/` which is outside the project directory and not served by any HTTP handler. |
| T-06-05 | Spoofing | session.ts — forged sessionId in logs | accept | `sessionId` is generated server-side via `crypto.randomUUID()` in `index.ts` — browser cannot control it. |
</threat_model>
